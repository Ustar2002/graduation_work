The factor is computed column by column (left-looking): the diagonal element of column j comes from row j alone, and the elements below it are independent dot products of row i with row j, so the rows below j are split off with `split_at_mut` and processed in parallel with `par_chunks_mut`, each task writing only its own row. Dot products are accumulated in f64 with a fixed summation order, and the part above the diagonal is zeroed first.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

/// Dot product of two equally long f32 slices, accumulated in f64 in a fixed order.
fn dot_f64(a: &[f32], b: &[f32]) -> f64 {
    let ca = a.chunks_exact(8);
    let cb = b.chunks_exact(8);
    let (ra, rb) = (ca.remainder(), cb.remainder());
    let mut acc = [0.0f64; 8];
    for (x, y) in ca.zip(cb) {
        for l in 0..8 {
            acc[l] += x[l] as f64 * y[l] as f64;
        }
    }
    let mut s = 0.0f64;
    for (x, y) in ra.iter().zip(rb.iter()) {
        s += *x as f64 * *y as f64;
    }
    s + ((acc[0] + acc[4]) + (acc[1] + acc[5])) + ((acc[2] + acc[6]) + (acc[3] + acc[7]))
}

#[no_mangle]
pub unsafe extern "C" fn cholesky(A: *const f32, L: *mut f32, n: i32) {
    if n <= 0 {
        return;
    }
    let n = n as usize;
    let a = std::slice::from_raw_parts(A, n * n);
    let l = std::slice::from_raw_parts_mut(L, n * n);

    // Everything above the diagonal is 0.
    l.par_chunks_mut(n).enumerate().for_each(|(i, row)| {
        for v in &mut row[i + 1..] {
            *v = 0.0;
        }
    });

    for j in 0..n {
        // Rows 0..=j are finished or being finished here; rows j+1.. are updated in parallel.
        let (top, bottom) = l.split_at_mut((j + 1) * n);
        let rowj = &mut top[j * n..(j + 1) * n];
        let d = a[j * n + j] as f64 - dot_f64(&rowj[..j], &rowj[..j]);
        let djj = d.sqrt();
        rowj[j] = djj as f32;
        let inv = 1.0 / djj;
        let rowj: &[f32] = &top[j * n..j * n + j];
        bottom.par_chunks_mut(n).enumerate().for_each(|(t, row)| {
            let i = j + 1 + t;
            let s = a[i * n + j] as f64 - dot_f64(&row[..j], rowj);
            row[j] = (s * inv) as f32;
        });
    }
}
```
