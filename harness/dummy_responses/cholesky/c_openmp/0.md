Right-looking (outer-product) Cholesky. The lower triangle of A is first copied into L and the upper triangle zeroed, in parallel over rows. Then for each column k the pivot column is scaled and the trailing submatrix is updated with a rank-1 term; the rows of the trailing update are independent, so that loop is parallelized, with the pivot column kept in a contiguous scratch vector so every inner loop runs along rows (contiguous in row-major) and can be vectorized.

```c
#include <math.h>
#include <stdlib.h>

void cholesky(const float* A, float* L, int n) {
    float* col = (float*)malloc((size_t)n * sizeof(float));

#pragma omp parallel for schedule(static)
    for (int i = 0; i < n; i++) {
        const float* Ai = A + (long)i * n;
        float* Li = L + (long)i * n;
        for (int j = 0; j <= i; j++) Li[j] = Ai[j];
        for (int j = i + 1; j < n; j++) Li[j] = 0.0f;
    }

    for (int k = 0; k < n; k++) {
        const float d = sqrtf(L[(long)k * n + k]);
        const float inv = 1.0f / d;
        L[(long)k * n + k] = d;
        for (int i = k + 1; i < n; i++) {
            const float v = L[(long)i * n + k] * inv;
            L[(long)i * n + k] = v;
            col[i] = v;
        }
#pragma omp parallel for schedule(dynamic, 16)
        for (int i = k + 1; i < n; i++) {
            float* Li = L + (long)i * n;
            const float lik = col[i];
            for (int j = k + 1; j <= i; j++) Li[j] -= lik * col[j];
        }
    }

    free(col);
}
```
