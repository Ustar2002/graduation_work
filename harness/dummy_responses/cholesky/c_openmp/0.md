The factor is built as the upper triangle R (A = R^T R) inside the output buffer, in blocks of rows. Once a block of rows is final, the rows below it are independent of each other, so that update loop is parallel. R is then mirrored below the diagonal and the upper part is cleared.

```c
#include <math.h>
#include <stddef.h>

void cholesky(const float* A, float* L, int n) {
    enum { NB = 32 }; /* 한 번에 확정하는 행의 수 */

    /* A 의 상삼각을 L 로 복사한다(대각 아래는 마지막 단계에서 전부 덮어쓴다) */
    for (int i = 0; i < n; i++) {
        const float *a = A + (size_t)i * n;
        float *r = L + (size_t)i * n;
        for (int j = i; j < n; j++) r[j] = a[j];
    }

    /* 상삼각 R 을 제자리에서 만든다. 블록 루프는 순차다(블록 kb 는 앞 블록들이 뺀 결과를 읽는다).
       원소 (i, j) 에서 빼는 순서는 블록·스레드 수와 무관하게 k = 0, 1, ..., i-1 이라 결과가 실행마다 같다. */
    for (int kb = 0; kb < n; kb += NB) {
        const int ke = kb + NB < n ? kb + NB : n;
        /* (1) 행 kb..ke-1 을 확정한다(순차): 블록 안의 앞 행을 읽으므로 행 사이에 의존이 있다 */
        for (int i = kb; i < ke; i++) {
            float *restrict ri = L + (size_t)i * n;
            for (int k = kb; k < i; k++) {
                const float *rk = L + (size_t)k * n;
                const float f = rk[i];
                for (int j = i; j < n; j++) ri[j] -= f * rk[j];
            }
            const float d = sqrtf(ri[i]);
            ri[i] = d;
            for (int j = i + 1; j < n; j++) ri[j] /= d;
        }
        /* (2) 확정한 행 kb..ke-1 의 기여를 그 아래 모든 행에서 뺀다(병렬).
           반복 i 는 행 i 만 쓰고 행 kb..ke-1 만 읽는다. 그 행들은 이 루프에서 쓰이지 않는다(i >= ke).
           삼각형이라 행마다 일의 양이 달라 조각을 번갈아 나눈다. */
#pragma omp parallel for schedule(static, 8)
        for (int i = ke; i < n; i++) {
            float *restrict ri = L + (size_t)i * n;
            for (int k = kb; k < ke; k++) {
                const float *rk = L + (size_t)k * n;
                const float f = rk[i];
                for (int j = i; j < n; j++) ri[j] -= f * rk[j];
            }
        }
    }

    /* L = R^T: 대각 위 원소를 대각 아래로 옮기고 대각 위는 0 으로 채운다 */
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            L[(size_t)j * n + i] = L[(size_t)i * n + j];
            L[(size_t)i * n + j] = 0.0f;
        }
    }
}
```
