The factor is built as the upper triangle R (A = R^T R) inside the output buffer, so every update runs along contiguous rows, and the rows are processed in blocks to keep the active rows in cache. R is then mirrored below the diagonal and the upper part is cleared.

```c
#include <math.h>
#include <stddef.h>

void cholesky(const float* A, float* L, int n) {
    enum { NB = 32 }; /* 한 번에 확정하는 행의 수 */

    /* A 의 상삼각을 L 로 복사한다(대각 아래는 마지막 단계에서 전부 덮어쓴다) */
    for (int i = 0; i < n; i++) {
        const float *a = A + (size_t)i * n;
        float *r = L + (size_t)i * n;
        for (int j = i; j < n; j++) r[j] = a[j];
    }

    /* 상삼각 R 을 제자리에서 만든다. 원소 (i, j) 에서 빼는 순서는 블록과 무관하게 k = 0, 1, ..., i-1 이다. */
    for (int kb = 0; kb < n; kb += NB) {
        const int ke = kb + NB < n ? kb + NB : n;
        /* (1) 행 kb..ke-1 을 확정한다: 앞 블록들의 기여는 이미 빠져 있고, 블록 안의 앞 행만 더 뺀다 */
        for (int i = kb; i < ke; i++) {
            float *restrict ri = L + (size_t)i * n;
            for (int k = kb; k < i; k++) {
                const float *rk = L + (size_t)k * n;
                const float f = rk[i];
                for (int j = i; j < n; j++) ri[j] -= f * rk[j];
            }
            const float d = sqrtf(ri[i]);
            ri[i] = d;
            for (int j = i + 1; j < n; j++) ri[j] /= d;
        }
        /* (2) 확정한 행 kb..ke-1 의 기여를 그 아래 모든 행에서 뺀다 */
        for (int i = ke; i < n; i++) {
            float *restrict ri = L + (size_t)i * n;
            for (int k = kb; k < ke; k++) {
                const float *rk = L + (size_t)k * n;
                const float f = rk[i];
                for (int j = i; j < n; j++) ri[j] -= f * rk[j];
            }
        }
    }

    /* L = R^T: 대각 위 원소를 대각 아래로 옮기고 대각 위는 0 으로 채운다 */
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            L[(size_t)j * n + i] = L[(size_t)i * n + j];
            L[(size_t)i * n + j] = 0.0f;
        }
    }
}
```
