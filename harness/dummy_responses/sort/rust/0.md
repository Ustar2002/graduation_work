The keys are copied into the output buffer, which is then sorted in place with rayon's parallel unstable sort; equal integers are indistinguishable, so the result is deterministic.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

#[no_mangle]
pub unsafe extern "C" fn sort(keys: *const i32, out: *mut i32, n: i32) {
    if n <= 0 {
        return;
    }
    let len = n as usize;
    let keys = std::slice::from_raw_parts(keys, len);
    let out = std::slice::from_raw_parts_mut(out, len);

    out.copy_from_slice(keys);
    // The parallel sort works only on out; keys is never written.
    out.par_sort_unstable();
}
```
