A four-pass byte-wise radix sort; flipping the sign bit makes the unsigned digit order equal to the signed order.

```c
#include <stdint.h>
#include <stdlib.h>
#include <string.h>

static int cmp_int(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);
}

void sort(const int* keys, int* out, int n) {
    size_t count[4][256];
    uint32_t *tmp = malloc((size_t)n * sizeof *tmp);
    uint32_t *dst = (uint32_t *)out;
    if (!tmp) { /* 작업 버퍼가 없으면 출력 버퍼에서 비교 정렬 */
        memcpy(out, keys, (size_t)n * sizeof *out);
        qsort(out, (size_t)n, sizeof *out, cmp_int);
        return;
    }
    memset(count, 0, sizeof count);
    for (int i = 0; i < n; i++) {
        uint32_t k = (uint32_t)keys[i] ^ 0x80000000u;
        count[0][k & 255]++;
        count[1][(k >> 8) & 255]++;
        count[2][(k >> 16) & 255]++;
        count[3][k >> 24]++;
    }
    for (int p = 0; p < 4; p++) { /* 개수 → 시작 위치 */
        size_t start = 0;
        for (int b = 0; b < 256; b++) {
            size_t c = count[p][b];
            count[p][b] = start;
            start += c;
        }
    }
    /* keys → tmp → out → tmp → out: 자릿수마다 안정적으로 옮긴다. 마지막에 부호 비트를 되돌린다 */
    for (int i = 0; i < n; i++) {
        uint32_t k = (uint32_t)keys[i] ^ 0x80000000u;
        tmp[count[0][k & 255]++] = k;
    }
    for (int i = 0; i < n; i++) {
        uint32_t k = tmp[i];
        dst[count[1][(k >> 8) & 255]++] = k;
    }
    for (int i = 0; i < n; i++) {
        uint32_t k = dst[i];
        tmp[count[2][(k >> 16) & 255]++] = k;
    }
    for (int i = 0; i < n; i++) {
        uint32_t k = tmp[i];
        dst[count[3][k >> 24]++] = k ^ 0x80000000u;
    }
    free(tmp);
}
```
