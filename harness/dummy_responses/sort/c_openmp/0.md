A least-significant-digit radix sort with four 8-bit passes, parallelised over contiguous chunks of the input. In each pass every thread builds a digit histogram of its own chunk; one thread then turns the per-thread histograms into write offsets ordered by bucket and, within a bucket, by thread, so the following scatter preserves the input order inside every bucket (the stability LSD radix sort needs); then every thread scatters its chunk. Keys travel with the sign bit flipped so that unsigned digit order equals signed order, the flip is undone in the last pass, and with four passes the last pass lands in `out`, so every element of `out` is written and `x` is only read.

```c
#include <stdlib.h>
#include <string.h>
#include <omp.h>

#define RADIX_BITS 8
#define RADIX (1u << RADIX_BITS)
#define RADIX_MASK (RADIX - 1u)
#define PASSES 4                 /* 4 x 8 bits cover every bit of a 32-bit key */
#define SIGN_BIT 0x80000000u

static int cmp_int(const void *a, const void *b)
{
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);
}

void sort(const int* x, int* out, int n)
{
    if (n <= 0)
        return;
    const size_t count = (size_t)n;
    unsigned *tmp = malloc(count * sizeof *tmp);
    const int max_threads = omp_get_max_threads();
    size_t *hist = malloc((size_t)max_threads * RADIX * sizeof *hist);
    if (!tmp || !hist) {
        /* out of memory: fall back to a sequential library sort of a copy */
        free(tmp);
        free(hist);
        memcpy(out, x, count * sizeof *out);
        qsort(out, count, sizeof *out, cmp_int);
        return;
    }
    const unsigned *input = (const unsigned *)x;
    /* pass p writes bufs[p & 1]; with an even number of passes the last pass lands in out */
    unsigned *bufs[2] = { tmp, (unsigned *)out };

#pragma omp parallel
    {
        const int T = omp_get_num_threads(), t = omp_get_thread_num();
        const size_t lo = count * (size_t)t / (size_t)T;
        const size_t hi = count * (size_t)(t + 1) / (size_t)T;
        size_t *h = hist + (size_t)t * RADIX;

        for (int pass = 0; pass < PASSES; pass++) {
            const unsigned shift = (unsigned)pass * RADIX_BITS;
            /* keys are kept with the sign bit flipped so unsigned digit order equals signed order:
               flipped on the way in (first pass), flipped back on the way out (last pass) */
            const unsigned flip_in = pass == 0 ? SIGN_BIT : 0u;
            const unsigned flip_out = pass == PASSES - 1 ? SIGN_BIT : 0u;
            const unsigned *src = pass == 0 ? input : bufs[(pass - 1) & 1];
            unsigned *dst = bufs[pass & 1];

            /* 1. digit histogram of this thread's chunk */
            memset(h, 0, RADIX * sizeof *h);
            for (size_t i = lo; i < hi; i++)
                h[((src[i] ^ flip_in) >> shift) & RADIX_MASK]++;
#pragma omp barrier

            /* 2. histograms -> write offsets, bucket-major and then in thread order: thread t's run of
                  bucket b starts after all smaller buckets and after bucket b of the threads before it,
                  so the scatter keeps the input order within a bucket (stable) */
#pragma omp single
            {
                size_t sum = 0;
                for (unsigned b = 0; b < RADIX; b++)
                    for (int u = 0; u < T; u++) {
                        size_t *cell = hist + (size_t)u * RADIX + b;
                        const size_t c = *cell;
                        *cell = sum;
                        sum += c;
                    }
            }   /* implicit barrier */

            /* 3. stable scatter of this thread's chunk */
            for (size_t i = lo; i < hi; i++) {
                const unsigned k = src[i] ^ flip_in;
                dst[h[(k >> shift) & RADIX_MASK]++] = k ^ flip_out;
            }
#pragma omp barrier
        }
    }
    free(tmp);
    free(hist);
}
```
