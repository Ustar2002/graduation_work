One chunk per thread is sorted in parallel, then neighbouring sorted runs are merged pairwise between two buffers.

```c
#include <omp.h>
#include <stdlib.h>
#include <string.h>

static int cmp_int(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);
}

/* 정렬된 두 구간 src[lo, mid), src[mid, hi) 를 dst[lo, hi) 로 병합한다(mid == hi 면 복사) */
static void merge_runs(const int *src, int *dst, size_t lo, size_t mid, size_t hi) {
    size_t i = lo, j = mid, k = lo;
    while (i < mid && j < hi) dst[k++] = (src[j] < src[i]) ? src[j++] : src[i++];
    while (i < mid) dst[k++] = src[i++];
    while (j < hi) dst[k++] = src[j++];
}

void sort(const int* keys, int* out, int n) {
    int chunks = omp_get_max_threads();
    if (chunks < 1) chunks = 1;
    if (chunks > n) chunks = n;
    size_t *bound = malloc(((size_t)chunks + 1) * sizeof *bound);
    int *tmp = malloc((size_t)n * sizeof *tmp);
    memcpy(out, keys, (size_t)n * sizeof *out);
    if (!bound || !tmp) { /* 작업 버퍼가 없으면 출력 버퍼에서 한 번에 정렬 */
        qsort(out, (size_t)n, sizeof *out, cmp_int);
        free(bound);
        free(tmp);
        return;
    }
    for (int c = 0; c <= chunks; c++) bound[c] = (size_t)n * (size_t)c / (size_t)chunks;

    /* 조각 c 는 out[bound[c], bound[c+1]) 만 읽고 쓴다: 반복끼리 겹치는 원소가 없다 */
#pragma omp parallel for schedule(static, 1)
    for (int c = 0; c < chunks; c++)
        qsort(out + bound[c], bound[c + 1] - bound[c], sizeof *out, cmp_int);

    /* 폭을 두 배씩 늘리며 이웃한 두 구간을 병합한다. 한 라운드의 반복 g 는 src 의 [bound[first], bound[last]) 를 읽고
       dst 의 같은 구간에만 쓴다(구간이 서로 겹치지 않고 src 와 dst 는 다른 버퍼다). 버퍼 교환은 병렬 영역 밖에서 한다 */
    int *src = out, *dst = tmp;
    for (int width = 1; width < chunks; width *= 2) {
        int groups = (chunks + 2 * width - 1) / (2 * width);
#pragma omp parallel for schedule(static, 1)
        for (int g = 0; g < groups; g++) {
            int first = g * 2 * width;
            int middle = first + width < chunks ? first + width : chunks;
            int last = first + 2 * width < chunks ? first + 2 * width : chunks;
            merge_runs(src, dst, bound[first], bound[middle], bound[last]);
        }
        int *swap = src;
        src = dst;
        dst = swap;
    }
    if (src != out) memcpy(out, src, (size_t)n * sizeof *out);
    free(tmp);
    free(bound);
}
```
