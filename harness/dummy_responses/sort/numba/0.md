The keys are copied into `out` and split into one contiguous segment per thread; the segments are sorted in parallel, and then neighbouring sorted runs are merged pairwise, in parallel, between `out` and a temporary buffer until a single sorted run remains in `out`.

```python
import numba
import numpy as np
from numba import njit, prange


@njit("void(int32[::1], int32[::1], int64, int64, int64)")
def merge_runs(src, dst, lo, mid, hi):
    # Merge the sorted runs src[lo:mid] and src[mid:hi] into dst[lo:hi].
    i = lo
    j = mid
    k = lo
    while i < mid and j < hi:
        if src[i] <= src[j]:
            dst[k] = src[i]
            i += 1
        else:
            dst[k] = src[j]
            j += 1
        k += 1
    while i < mid:
        dst[k] = src[i]
        i += 1
        k += 1
    while j < hi:
        dst[k] = src[j]
        j += 1
        k += 1


@njit("void(int32[::1], int32[::1], int64)", parallel=True)
def sort(keys, out, n):
    nseg = numba.get_num_threads()
    if nseg > n:
        nseg = n
    if nseg < 1:
        nseg = 1
    seg = (n + nseg - 1) // nseg
    for i in prange(n):
        out[i] = keys[i]
    # Iteration s sorts only its own segment out[lo:hi].
    for s in prange(nseg):
        lo = min(s * seg, n)
        hi = min(lo + seg, n)
        out[lo:hi].sort()
    tmp = np.empty(n, dtype=np.int32)
    src = out
    dst = tmp
    in_out = True
    width = 1
    while width < nseg:
        npairs = (nseg + 2 * width - 1) // (2 * width)
        # Iteration p reads src[lo:hi] and writes only dst[lo:hi]; the ranges are disjoint.
        for p in prange(npairs):
            lo = min(p * 2 * width * seg, n)
            mid = min(lo + width * seg, n)
            hi = min(lo + 2 * width * seg, n)
            merge_runs(src, dst, lo, mid, hi)
        src, dst = dst, src
        in_out = not in_out
        width *= 2
    if not in_out:
        for i in prange(n):
            out[i] = tmp[i]
```
