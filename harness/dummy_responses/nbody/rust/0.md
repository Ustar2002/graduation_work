The positions are first copied into three separate coordinate arrays; then every body's acceleration is summed over all bodies in parallel, one output row per task, with eight independent accumulators per coordinate so the inner loop vectorizes.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

const EPS2: f32 = 0.0001;
const LANES: usize = 8;

#[inline(always)]
fn term(xj: f32, yj: f32, zj: f32, mj: f32, xi: f32, yi: f32, zi: f32) -> (f32, f32, f32) {
    let dx = xj - xi;
    let dy = yj - yi;
    let dz = zj - zi;
    let r2 = dx * dx + dy * dy + dz * dz + EPS2;
    let inv = 1.0 / r2.sqrt();
    let s = mj * inv * inv * inv;
    (s * dx, s * dy, s * dz)
}

#[no_mangle]
pub unsafe extern "C" fn nbody(pos: *const f32, mass: *const f32, acc: *mut f32, n: i32) {
    if n <= 0 {
        return;
    }
    let len = n as usize;
    let pos = std::slice::from_raw_parts(pos, 3 * len);
    let mass = std::slice::from_raw_parts(mass, len);
    let acc = std::slice::from_raw_parts_mut(acc, 3 * len);

    // Structure-of-arrays copy of the positions (read-only below).
    let px: Vec<f32> = pos.chunks_exact(3).map(|p| p[0]).collect();
    let py: Vec<f32> = pos.chunks_exact(3).map(|p| p[1]).collect();
    let pz: Vec<f32> = pos.chunks_exact(3).map(|p| p[2]).collect();

    // Each closure writes only its own row acc[3i..3i+3]: disjoint chunks, no races.
    acc.par_chunks_mut(3).enumerate().for_each(|(i, row)| {
        let (xi, yi, zi) = (px[i], py[i], pz[i]);
        let mut ax = [0.0f32; LANES];
        let mut ay = [0.0f32; LANES];
        let mut az = [0.0f32; LANES];
        // Blocks of LANES bodies: fixed-size chunks, so the lane loop has no bounds checks.
        let blocks = px
            .chunks_exact(LANES)
            .zip(py.chunks_exact(LANES))
            .zip(pz.chunks_exact(LANES))
            .zip(mass.chunks_exact(LANES));
        for (((bx, by), bz), bm) in blocks {
            for l in 0..LANES {
                let (tx, ty, tz) = term(bx[l], by[l], bz[l], bm[l], xi, yi, zi);
                ax[l] += tx;
                ay[l] += ty;
                az[l] += tz;
            }
        }
        let mut sx: f32 = ax.iter().sum();
        let mut sy: f32 = ay.iter().sum();
        let mut sz: f32 = az.iter().sum();
        // Remaining len % LANES bodies.
        for j in len / LANES * LANES..len {
            let (tx, ty, tz) = term(px[j], py[j], pz[j], mass[j], xi, yi, zi);
            sx += tx;
            sy += ty;
            sz += tz;
        }
        row[0] = sx;
        row[1] = sy;
        row[2] = sz;
    });
}
```
