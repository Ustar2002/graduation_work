The positions are first copied into three separate coordinate arrays; then every particle's acceleration is summed over all particles in parallel, one output row per task, with eight independent accumulators per coordinate so the inner loop vectorizes.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

const EPS2: f32 = 0.0001;
const LANES: usize = 8;

#[no_mangle]
pub unsafe extern "C" fn nbody(pos: *const f32, mass: *const f32, acc: *mut f32, n: i32) {
    if n <= 0 {
        return;
    }
    let len = n as usize;
    let pos = std::slice::from_raw_parts(pos, 3 * len);
    let mass = std::slice::from_raw_parts(mass, len);
    let acc = std::slice::from_raw_parts_mut(acc, 3 * len);

    // Structure-of-arrays copy of the positions (read-only below).
    let px: Vec<f32> = (0..len).map(|j| pos[3 * j]).collect();
    let py: Vec<f32> = (0..len).map(|j| pos[3 * j + 1]).collect();
    let pz: Vec<f32> = (0..len).map(|j| pos[3 * j + 2]).collect();

    // Each closure writes only its own row acc[3i..3i+3]: disjoint chunks, no races.
    acc.par_chunks_mut(3).enumerate().for_each(|(i, row)| {
        let (xi, yi, zi) = (px[i], py[i], pz[i]);
        let mut ax = [0.0f32; LANES];
        let mut ay = [0.0f32; LANES];
        let mut az = [0.0f32; LANES];
        let full = len / LANES * LANES;
        let mut j = 0;
        while j < full {
            for l in 0..LANES {
                let dx = px[j + l] - xi;
                let dy = py[j + l] - yi;
                let dz = pz[j + l] - zi;
                let r2 = dx * dx + dy * dy + dz * dz + EPS2;
                let inv = 1.0 / r2.sqrt();
                let s = mass[j + l] * inv * inv * inv;
                ax[l] += s * dx;
                ay[l] += s * dy;
                az[l] += s * dz;
            }
            j += LANES;
        }
        let mut sx: f32 = ax.iter().sum();
        let mut sy: f32 = ay.iter().sum();
        let mut sz: f32 = az.iter().sum();
        for j in full..len {
            let dx = px[j] - xi;
            let dy = py[j] - yi;
            let dz = pz[j] - zi;
            let r2 = dx * dx + dy * dy + dz * dz + EPS2;
            let inv = 1.0 / r2.sqrt();
            let s = mass[j] * inv * inv * inv;
            sx += s * dx;
            sy += s * dy;
            sz += s * dz;
        }
        row[0] = sx;
        row[1] = sy;
        row[2] = sz;
    });
}
```
