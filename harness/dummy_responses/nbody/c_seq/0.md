A direct double loop over all pairs; the j = i term is zero, so it is not skipped.

```c
#include <math.h>
#include <stddef.h>

void nbody(const float* pos, const float* mass, float* acc, int n) {
    const float eps2 = 0.01f * 0.01f;
    for (int i = 0; i < n; i++) {
        const float xi = pos[3 * (size_t)i], yi = pos[3 * (size_t)i + 1], zi = pos[3 * (size_t)i + 2];
        float ax = 0.0f, ay = 0.0f, az = 0.0f;
        for (int j = 0; j < n; j++) {
            float dx = pos[3 * (size_t)j] - xi;
            float dy = pos[3 * (size_t)j + 1] - yi;
            float dz = pos[3 * (size_t)j + 2] - zi;
            float r2 = dx * dx + dy * dy + dz * dz + eps2;
            float inv = 1.0f / sqrtf(r2);
            float s = mass[j] * (inv * inv * inv);
            ax += s * dx;
            ay += s * dy;
            az += s * dz;
        }
        acc[3 * (size_t)i] = ax;
        acc[3 * (size_t)i + 1] = ay;
        acc[3 * (size_t)i + 2] = az;
    }
}
```
