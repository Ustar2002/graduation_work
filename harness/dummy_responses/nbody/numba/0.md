The positions are first copied into three separate double-precision arrays; then each particle's acceleration is accumulated over all particles in double precision in a parallel loop over the particles, where iteration i writes only row i of `acc`.

```python
import numpy as np
from numba import njit, prange


@njit("void(float32[:, ::1], float32[::1], float32[:, ::1], int64)", parallel=True)
def nbody(pos, mass, acc, n):
    eps2 = 0.0001
    px = np.empty(n, dtype=np.float64)
    py = np.empty(n, dtype=np.float64)
    pz = np.empty(n, dtype=np.float64)
    m = np.empty(n, dtype=np.float64)
    for j in prange(n):
        px[j] = pos[j, 0]
        py[j] = pos[j, 1]
        pz[j] = pos[j, 2]
        m[j] = mass[j]
    for i in prange(n):
        xi = px[i]
        yi = py[i]
        zi = pz[i]
        ax = 0.0
        ay = 0.0
        az = 0.0
        for j in range(n):
            dx = px[j] - xi
            dy = py[j] - yi
            dz = pz[j] - zi
            r2 = dx * dx + dy * dy + dz * dz + eps2
            inv = 1.0 / np.sqrt(r2)
            w = m[j] * inv * inv * inv
            ax += w * dx
            ay += w * dy
            az += w * dz
        acc[i, 0] = ax
        acc[i, 1] = ay
        acc[i, 2] = az
```
