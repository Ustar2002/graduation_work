The sum over all bodies j is a reduction domain; blocks of 8 bodies i are vectorized and distributed over the threads, and each block accumulates only its own three components, so the result does not depend on the thread schedule.

```python
import halide as hl


def nbody(pos, mass, n):
    eps2 = hl.f32(0.0001)
    i, d = hl.Var("i"), hl.Var("d")
    j = hl.RDom([(0, n)], "j")

    dx = pos[0, j.x] - pos[0, i]
    dy = pos[1, j.x] - pos[1, i]
    dz = pos[2, j.x] - pos[2, i]
    r2 = dx * dx + dy * dy + dz * dz + eps2
    w = mass[j.x] / (r2 * hl.sqrt(r2))

    # s[i, d]: component d of the acceleration of body i
    s = hl.Func("s")
    s[i, d] = 0.0
    s[i, d] += w * (pos[d, j.x] - pos[d, i])

    acc = hl.Func("acc")
    acc[d, i] = s[i, d]

    io, ii = hl.Var("io"), hl.Var("ii")
    s.compute_root()
    s.update(0).split(i, io, ii, 8, hl.TailStrategy.GuardWithIf).reorder(j.x, ii, d, io).vectorize(ii).parallel(io)
    return acc
```
