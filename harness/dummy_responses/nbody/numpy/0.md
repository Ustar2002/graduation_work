Rows are processed in blocks, so the temporaries are (block, n) arrays instead of n*n*3 values.

```python
import numpy as np

def nbody(pos, mass, acc, n):
    eps2 = np.float32(0.01 * 0.01)
    px = np.ascontiguousarray(pos[:, 0])
    py = np.ascontiguousarray(pos[:, 1])
    pz = np.ascontiguousarray(pos[:, 2])
    block = max(1, min(n, 131072 // n))
    for i0 in range(0, n, block):
        i1 = min(n, i0 + block)
        dx = px[None, :] - px[i0:i1, None]
        dy = py[None, :] - py[i0:i1, None]
        dz = pz[None, :] - pz[i0:i1, None]
        r2 = dx * dx + dy * dy + dz * dz + eps2
        w = mass[None, :] / (r2 * np.sqrt(r2))
        acc[i0:i1, 0] = np.sum(w * dx, axis=1)
        acc[i0:i1, 1] = np.sum(w * dy, axis=1)
        acc[i0:i1, 2] = np.sum(w * dz, axis=1)
```
