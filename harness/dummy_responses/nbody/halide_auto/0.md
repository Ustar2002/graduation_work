The sum over all particles j is a reduction domain, and every particle i accumulates only its own three components.

```python
import halide as hl


def nbody(pos, mass, n):
    eps2 = hl.f32(0.0001)
    i, d = hl.Var("i"), hl.Var("d")
    j = hl.RDom([(0, n)], "j")

    dx = pos[0, j.x] - pos[0, i]
    dy = pos[1, j.x] - pos[1, i]
    dz = pos[2, j.x] - pos[2, i]
    r2 = dx * dx + dy * dy + dz * dz + eps2
    w = mass[j.x] / (r2 * hl.sqrt(r2))

    # s[i, d]: component d of the acceleration of particle i
    s = hl.Func("s")
    s[i, d] = 0.0
    s[i, d] += w * (pos[d, j.x] - pos[d, i])

    acc = hl.Func("acc")
    acc[d, i] = s[i, d]

    return acc
```
