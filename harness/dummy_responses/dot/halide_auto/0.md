The vector is cut into chunks of 16384 elements: each chunk is summed in double precision by its own reduction, then the chunk sums and the short tail are added together in a fixed order, so the result is deterministic and the scheduler can run the chunks in parallel.

```python
import halide as hl

CHUNK = 1 << 14


def dot(x, y, n):
    # The sizes are Python ints, so the pipeline is shaped for this n: full chunks, then the tail.
    n_full = n // CHUNK
    tail = n - n_full * CHUNK
    i = hl.Var("i")
    terms = []

    if n_full > 0:
        c = hl.Var("c")
        r = hl.RDom([(0, CHUNK)], "r")
        part = hl.Func("part")
        part[c] = hl.f64(0)
        part[c] += hl.f64(x[c * CHUNK + r.x]) * hl.f64(y[c * CHUNK + r.x])
        rc = hl.RDom([(0, n_full)], "rc")
        chunks = hl.Func("chunks")
        chunks[i] = hl.f64(0)
        chunks[i] += part[rc.x]
        terms.append(chunks)

    if tail > 0:
        rt = hl.RDom([(n_full * CHUNK, tail)], "rt")
        rest = hl.Func("rest")
        rest[i] = hl.f64(0)
        rest[i] += hl.f64(x[rt.x]) * hl.f64(y[rt.x])
        terms.append(rest)

    total = terms[0][i] if len(terms) == 1 else terms[0][i] + terms[1][i]
    out = hl.Func("out")
    out[i] = hl.f32(total)
    return out
```
