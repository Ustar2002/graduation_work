The vectors are cut into fixed-size blocks that are summed in parallel in double precision; the partial sums are then added in block order, so the result does not depend on the number of threads.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

#[no_mangle]
pub unsafe extern "C" fn dot(x: *const f32, y: *const f32, out: *mut f32, n: i32) {
    let out = std::slice::from_raw_parts_mut(out, 1);
    if n <= 0 {
        out[0] = 0.0;
        return;
    }
    let len = n as usize;
    let x = std::slice::from_raw_parts(x, len);
    let y = std::slice::from_raw_parts(y, len);

    const BLOCK: usize = 1 << 16;
    // Each closure only reads x and y and returns its own partial sum: no shared writes.
    let partials: Vec<f64> = x
        .par_chunks(BLOCK)
        .zip(y.par_chunks(BLOCK))
        .map(|(xb, yb)| {
            // Eight independent accumulators let the compiler vectorize the loop.
            let mut acc = [0.0f64; 8];
            let mut xc = xb.chunks_exact(8);
            let mut yc = yb.chunks_exact(8);
            for (xs, ys) in (&mut xc).zip(&mut yc) {
                for l in 0..8 {
                    acc[l] += xs[l] as f64 * ys[l] as f64;
                }
            }
            let mut s: f64 = acc.iter().sum();
            for (&a, &b) in xc.remainder().iter().zip(yc.remainder()) {
                s += a as f64 * b as f64;
            }
            s
        })
        .collect();
    out[0] = partials.iter().sum::<f64>() as f32;
}
```
