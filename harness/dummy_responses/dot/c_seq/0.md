Accumulation is done in double so the sum stays accurate for large n.

```c
void dot(const float* x, const float* y, float* out, int n) {
    double s0 = 0.0, s1 = 0.0, s2 = 0.0, s3 = 0.0;
    int i = 0;
    for (; i + 3 < n; i += 4) {
        s0 += (double)x[i] * y[i];
        s1 += (double)x[i + 1] * y[i + 1];
        s2 += (double)x[i + 2] * y[i + 2];
        s3 += (double)x[i + 3] * y[i + 3];
    }
    for (; i < n; i++) s0 += (double)x[i] * y[i];
    out[0] = (float)((s0 + s1) + (s2 + s3));
}
```
