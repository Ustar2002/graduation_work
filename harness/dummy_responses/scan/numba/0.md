This is a two-pass block scan: the array is split into one contiguous block per thread, the block totals are computed in parallel and scanned sequentially to get each block's starting offset, and a second parallel pass writes the running totals of each block, starting from its offset.

```python
import numba
import numpy as np
from numba import njit, prange


@njit("void(int32[::1], int32[::1], int64)", parallel=True)
def scan(x, out, n):
    nblocks = numba.get_num_threads()
    if nblocks > n:
        nblocks = n
    if nblocks < 1:
        nblocks = 1
    block = (n + nblocks - 1) // nblocks
    sums = np.zeros(nblocks, dtype=np.int64)
    # Pass 1: iteration b writes only sums[b].
    for b in prange(nblocks):
        start = b * block
        stop = min(start + block, n)
        s = 0
        for i in range(start, stop):
            s += x[i]
        sums[b] = s
    # Exclusive scan of the block totals (sequential, nblocks values).
    offsets = np.zeros(nblocks, dtype=np.int64)
    run = 0
    for b in range(nblocks):
        offsets[b] = run
        run += sums[b]
    # Pass 2: iteration b writes only out[start:stop] of its own block.
    for b in prange(nblocks):
        start = b * block
        stop = min(start + block, n)
        s = offsets[b]
        for i in range(start, stop):
            s += x[i]
            out[i] = s
```
