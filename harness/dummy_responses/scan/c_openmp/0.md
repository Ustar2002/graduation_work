Two passes over contiguous per-thread blocks: every thread first sums its own block, the block sums are turned into exclusive offsets, and then every thread scans its block sequentially starting from its offset. The input is read twice and the output written once, which is as little traffic as a parallel scan needs.

```c
#include <omp.h>

#define MAX_THREADS 256

void scan(const int* x, int* out, int n) {
    if (n <= 0) return;
    int nthreads = omp_get_max_threads();
    if (nthreads > MAX_THREADS) nthreads = MAX_THREADS;
    if (nthreads < 1) nthreads = 1;
    long long block_sum[MAX_THREADS];

#pragma omp parallel num_threads(nthreads)
    {
        const int nt = omp_get_num_threads();
        const int tid = omp_get_thread_num();
        const long long chunk = ((long long)n + nt - 1) / nt;
        long long lo = (long long)tid * chunk;
        long long hi = lo + chunk;
        if (lo > n) lo = n;
        if (hi > n) hi = n;

        /* pass 1: sum of this thread's block */
        long long s = 0;
        for (long long i = lo; i < hi; i++) s += x[i];
        block_sum[tid] = s;
#pragma omp barrier

        /* exclusive offset = sum of the blocks before this one */
        long long offset = 0;
        for (int t = 0; t < tid; t++) offset += block_sum[t];

        /* pass 2: local inclusive scan starting from the offset */
        int acc = (int)offset;
        for (long long i = lo; i < hi; i++) {
            acc += x[i];
            out[i] = acc;
        }
    }
}
```
