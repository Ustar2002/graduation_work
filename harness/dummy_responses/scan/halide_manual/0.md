The array is scanned in blocks of 8192 elements that are independent of each other; an exclusive scan of the block totals then gives each block the offset that is added to its local running totals.

```python
import halide as hl


def scan(x, n):
    block = 8192
    blocks = (n + block - 1) // block

    i, j, k = hl.Var("i"), hl.Var("j"), hl.Var("k")

    # element j of block k, zero past the end of x
    xb = hl.Func("xb")
    idx = k * block + j
    xb[j, k] = hl.select(idx < n, x[hl.min(idx, n - 1)], 0)

    # inclusive scan inside every block
    local = hl.Func("local")
    local[j, k] = xb[j, k]
    rj = hl.RDom([(1, block - 1)], "rj")
    local[rj.x, k] = local[rj.x - 1, k] + xb[rj.x, k]

    # offset[k] = sum of the totals of blocks 0 .. k-1
    offset = hl.Func("offset")
    offset[k] = 0
    rk = hl.RDom([(0, blocks)], "rk")
    offset[rk.x + 1] = offset[rk.x] + local[block - 1, rk.x]

    out = hl.Func("out")
    out[i] = local[i % block, i // block] + offset[i // block]

    io, ii = hl.Var("io"), hl.Var("ii")
    local.compute_root().vectorize(j, 8).parallel(k)
    local.update(0).parallel(k)
    offset.compute_root()
    out.split(i, io, ii, block, hl.TailStrategy.GuardWithIf).parallel(io).vectorize(ii, 8, hl.TailStrategy.GuardWithIf)
    return out
```
