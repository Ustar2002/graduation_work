This is a radix-2 Stockham transform: log2(n) out-of-place stages, each one Func of (real, imaginary) pairs, with the twiddle factors evaluated once in double precision; every stage is computed into its own buffer, split into blocks for the threads and vectorized inside a block.

```python
import math

import halide as hl


def fft(x, n):
    o, i = hl.Var("o"), hl.Var("i")
    # Twiddle factors w[p] = exp(-2*pi*i*p/n) for 0 <= p < n/2, evaluated in double precision.
    tw = hl.Func("tw")
    angle = hl.f64(-2.0 * math.pi / n) * hl.f64(o)
    tw[o] = (hl.f32(hl.cos(angle)), hl.f32(hl.sin(angle)))
    # Complex input element o as a (re, im) pair.
    X = hl.Func("X0")
    X[o] = (x[2 * o], x[2 * o + 1])
    # Stockham radix-2 decimation in frequency: log2(n) out-of-place stages, output in natural order.
    s, m = 1, n // 2  # stride and half length of the sub-transforms of this stage (2*m*s == n)
    t = 0
    stages = []
    while m >= 1:
        t += 1
        Y = hl.Func(f"X{t}")
        q = o % s
        p = (o // s) // 2
        e = (o // s) % 2
        a = X[q + s * p]
        b = X[q + s * (p + m)]
        w = tw[p * s]  # exp(-2*pi*i*p/(2*m))
        dr, di = a[0] - b[0], a[1] - b[1]
        Y[o] = (hl.select(e == 0, a[0] + b[0], dr * w[0] - di * w[1]),
                hl.select(e == 0, a[1] + b[1], dr * w[1] + di * w[0]))
        stages.append(Y)
        X = Y
        s, m = 2 * s, m // 2
    # Interleave the (re, im) pairs into the output.
    out = hl.Func("out")
    out[i] = hl.select(i % 2 == 0, X[i // 2][0], X[i // 2][1])

    # Schedule: the twiddle table and every stage are computed once into their own buffer (each element of a
    # stage is read twice by the next one); each is split into blocks of 4096 elements for the threads and
    # vectorized by 8 inside a block. Guarded tails keep every stage inside [0, n) for small n.
    g = hl.TailStrategy.GuardWithIf
    tw.compute_root().vectorize(o, 8, g).parallel(o, 4096 // 8, g)
    for f in stages:
        f.compute_root().vectorize(o, 8, g).parallel(o, 4096 // 8, g)
    out.vectorize(i, 8, g).parallel(i, 8192 // 8, g)
    return out
```
