This is an iterative radix-2 Cooley-Tukey FFT in double precision: the input is gathered in bit-reversed order, the twiddle factors are computed directly with `cos`/`sin`, and each stage runs its butterflies in parallel, over whole blocks with `par_chunks_mut` when there are many blocks and over matching chunks of the two halves of a block when there are few. Every task writes only its own disjoint chunks.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

type C64 = (f64, f64);

const CHUNK: usize = 4096;

/// Butterflies between lo[j] and hi[j]; j0 is the index of lo[0] within its half block.
fn butterflies(lo: &mut [C64], hi: &mut [C64], tw: &[C64], step: usize, j0: usize) {
    for (j, (a, b)) in lo.iter_mut().zip(hi.iter_mut()).enumerate() {
        let w = tw[(j0 + j) * step];
        let t = (b.0 * w.0 - b.1 * w.1, b.0 * w.1 + b.1 * w.0);
        *b = (a.0 - t.0, a.1 - t.1);
        *a = (a.0 + t.0, a.1 + t.1);
    }
}

#[no_mangle]
pub unsafe extern "C" fn fft(x: *const f32, out: *mut f32, n: i32) {
    if n <= 0 {
        return;
    }
    let n = n as usize;
    let xs = std::slice::from_raw_parts(x, 2 * n);
    let os = std::slice::from_raw_parts_mut(out, 2 * n);
    if n == 1 {
        os.copy_from_slice(xs);
        return;
    }

    let shift = usize::BITS - n.trailing_zeros();
    let mut buf: Vec<C64> = vec![(0.0, 0.0); n];
    buf.par_iter_mut().enumerate().for_each(|(i, z)| {
        let j = i.reverse_bits() >> shift;
        *z = (xs[2 * j] as f64, xs[2 * j + 1] as f64);
    });

    // tw[m] = exp(-2*pi*i*m/n) for 0 <= m < n/2
    let tw: Vec<C64> = (0..n / 2)
        .into_par_iter()
        .map(|m| {
            let t = -2.0 * std::f64::consts::PI * (m as f64) / (n as f64);
            (t.cos(), t.sin())
        })
        .collect();

    let mut len = 2;
    while len <= n {
        let half = len / 2;
        let step = n / len;
        if half <= CHUNK {
            buf.par_chunks_mut(len).for_each(|blk| {
                let (lo, hi) = blk.split_at_mut(half);
                butterflies(lo, hi, &tw, step, 0);
            });
        } else {
            for blk in buf.chunks_mut(len) {
                let (lo, hi) = blk.split_at_mut(half);
                lo.par_chunks_mut(CHUNK)
                    .zip(hi.par_chunks_mut(CHUNK))
                    .enumerate()
                    .for_each(|(c, (l, h))| butterflies(l, h, &tw, step, c * CHUNK));
            }
        }
        len *= 2;
    }

    os.par_chunks_mut(2).zip(buf.par_iter()).for_each(|(o, z)| {
        o[0] = z.0 as f32;
        o[1] = z.1 as f32;
    });
}
```
