An iterative radix-2 decimation-in-time transform: the input is copied into out in bit-reversed order, then log2(n) stages of butterflies run in place. Data stays float32; the twiddle factors of each stage are computed once per stage in double precision with cos/sin and the butterfly arithmetic is done in double before rounding back to float32.

```c
#include <math.h>
#include <stdlib.h>

void fft(const float* x, float* out, int n) {
    int logn = 0;
    while ((1 << logn) < n) logn++;
    /* copy the input into out in bit-reversed order */
    for (int i = 0; i < n; i++) {
        unsigned r = 0, v = (unsigned)i;
        for (int b = 0; b < logn; b++) { r = (r << 1) | (v & 1u); v >>= 1; }
        out[2 * r] = x[2 * i];
        out[2 * r + 1] = x[2 * i + 1];
    }
    if (n < 2) return;
    double* wr = (double*)malloc(sizeof(double) * (size_t)(n / 2));
    double* wi = (double*)malloc(sizeof(double) * (size_t)(n / 2));
    if (wr == NULL || wi == NULL) { free(wr); free(wi); return; }
    /* radix-2 stages; twiddles for each stage are computed in double precision */
    for (int len = 2; len <= n; len <<= 1) {
        const int half = len / 2;
        const double ang = -2.0 * M_PI / (double)len;
        for (int j = 0; j < half; j++) { wr[j] = cos(ang * j); wi[j] = sin(ang * j); }
        for (int start = 0; start < n; start += len) {
            for (int j = 0; j < half; j++) {
                const int a = 2 * (start + j), b = a + 2 * half;
                const double tr = wr[j] * out[b] - wi[j] * out[b + 1];
                const double ti = wr[j] * out[b + 1] + wi[j] * out[b];
                const double ur = out[a], ui = out[a + 1];
                out[a] = (float)(ur + tr);
                out[a + 1] = (float)(ui + ti);
                out[b] = (float)(ur - tr);
                out[b + 1] = (float)(ui - ti);
            }
        }
    }
    free(wr);
    free(wi);
}
```
