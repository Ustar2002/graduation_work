The input is copied into `out` in bit-reversed order and then combined in place by log2(n) passes of butterflies, which run one after another; inside a pass the threads share whole blocks while there are many blocks, and the butterflies of each block once only a few long blocks remain. No two iterations of a parallel loop touch the same element, so the result does not depend on the number of threads.

```c
#include <math.h>
#include <stdlib.h>

/* 블록의 k 번째 버터플라이: 앞 절반 a 의 복소수 k 와 뒤 절반 b 의 복소수 k 만 읽고 쓴다 */
static inline void butterfly(float *a, float *b, const float *tw, size_t step, int k) {
    const float wr = tw[2 * k * step], wi = tw[2 * k * step + 1];
    const float tr = b[2 * k] * wr - b[2 * k + 1] * wi;
    const float ti = b[2 * k] * wi + b[2 * k + 1] * wr;
    const float ar = a[2 * k], ai = a[2 * k + 1];
    a[2 * k] = ar + tr;
    a[2 * k + 1] = ai + ti;
    b[2 * k] = ar - tr;
    b[2 * k + 1] = ai - ti;
}

void fft(const float* x, float* out, int n) {
    enum { MIN_BLOCKS = 64 }; /* 블록이 이보다 적은 단계에서는 블록 안의 버터플라이를 나눈다 */
    int log2n = 0;
    while ((1 << log2n) < n) log2n++;

    /* 회전 인자 표: tw[2k], tw[2k+1] = exp(-2*pi*i*k/n) 의 실수부·허수부 (0 <= k < n/2). 배정밀도로 구해 float 로 저장한다.
       반복 k 는 자기 원소 둘만 쓴다 */
    float *tw = malloc((size_t)n * sizeof(float));
    if (!tw) return;
    const double base = -6.283185307179586476925286766559 / (double)n;
#pragma omp parallel for schedule(static)
    for (int k = 0; k < n / 2; k++) {
        tw[2 * (size_t)k] = (float)cos(base * k);
        tw[2 * (size_t)k + 1] = (float)sin(base * k);
    }

    /* 비트 반전 순서로 복사한다: out[rev(j)] = x[j]. rev(j) 를 반복마다 따로 계산하므로 반복 사이에 의존이 없고,
       rev 는 일대일 대응이라 서로 다른 j 가 같은 원소에 쓰지 않는다 */
#pragma omp parallel for schedule(static)
    for (int j = 0; j < n; j++) {
        unsigned r = 0;
        for (int bit = 0; bit < log2n; bit++) r |= (((unsigned)j >> bit) & 1u) << (log2n - 1 - bit);
        out[2 * (size_t)r] = x[2 * (size_t)j];
        out[2 * (size_t)r + 1] = x[2 * (size_t)j + 1];
    }

    /* 길이 2, 4, ..., n 의 블록을 차례로 합친다. 단계 s 는 단계 s-1 의 결과를 읽으므로 단계 루프는 순차다 */
    for (int s = 1; s <= log2n; s++) {
        const int len = 1 << s, half = len >> 1;
        const size_t step = (size_t)(n >> s); /* 이 단계의 블록 수이자, k 번째 회전 인자의 tw 안 간격 */
        if (n >> s >= MIN_BLOCKS) {
            /* 블록이 많다: 블록 단위로 나눈다. 블록 i 는 out 의 복소수 i .. i+len-1 만 읽고 쓴다 */
#pragma omp parallel for schedule(static)
            for (int i = 0; i < n; i += len) {
                float *a = out + 2 * (size_t)i;
                float *b = a + 2 * (size_t)half;
                for (int k = 0; k < half; k++) butterfly(a, b, tw, step, k);
            }
        } else {
            /* 긴 블록이 몇 개뿐이다: 블록마다 버터플라이를 나눈다. 반복 k 는 블록의 복소수 k 와 half+k 만 읽고 쓴다 */
            for (int i = 0; i < n; i += len) {
                float *a = out + 2 * (size_t)i;
                float *b = a + 2 * (size_t)half;
#pragma omp parallel for schedule(static)
                for (int k = 0; k < half; k++) butterfly(a, b, tw, step, k);
            }
        }
    }
    free(tw);
}
```
