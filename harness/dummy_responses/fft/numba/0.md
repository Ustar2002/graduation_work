Iterative radix-2 FFT in a complex128 work array: a bit-reversal permutation followed by log2(n) butterfly stages, with the twiddle factors computed directly in double precision. Within a stage every butterfly touches its own pair of elements, so the butterflies run in parallel with `prange`.

```python
import numpy as np
from numba import njit, prange


@njit("void(float32[::1], float32[::1], int64)", parallel=True)
def fft(x, out, n):
    bits = 0
    while (np.int64(1) << bits) < n:
        bits += 1

    half_n = n // 2
    tw = np.empty(half_n, dtype=np.complex128)
    for k in prange(half_n):
        t = -2.0 * np.pi * k / n
        tw[k] = complex(np.cos(t), np.sin(t))

    # Bit-reversal permutation: rev is a bijection, so every target is written once.
    a = np.empty(n, dtype=np.complex128)
    for i in prange(n):
        r = np.int64(0)
        v = np.int64(i)
        for b in range(bits):
            r = (r << 1) | (v & 1)
            v >>= 1
        a[r] = complex(np.float64(x[2 * i]), np.float64(x[2 * i + 1]))

    half = np.int64(1)
    while half < n:
        stride = n // (2 * half)
        for b in prange(half_n):
            pos = b % half
            i = (b // half) * (2 * half) + pos
            j = i + half
            t = tw[pos * stride] * a[j]
            u = a[i]
            a[i] = u + t
            a[j] = u - t
        half *= 2

    for k in prange(n):
        out[2 * k] = np.float32(a[k].real)
        out[2 * k + 1] = np.float32(a[k].imag)
```
