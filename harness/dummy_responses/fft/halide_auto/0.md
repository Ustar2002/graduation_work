This is a radix-2 Stockham transform: log2(n) out-of-place stages, each one Func of (real, imaginary) pairs, with the twiddle factors evaluated once in double precision by their own Func, so the result comes out in natural order without a bit-reversal pass; scheduling is left to the autoscheduler.

```python
import math

import halide as hl


def fft(x, n):
    o, c, k = hl.Var("o"), hl.Var("c"), hl.Var("k")
    # Twiddle factors w[p] = exp(-2*pi*i*p/n) for 0 <= p < n/2, evaluated in double precision.
    tw = hl.Func("tw")
    angle = hl.f64(-2.0 * math.pi / n) * hl.f64(o)
    tw[o] = (hl.f32(hl.cos(angle)), hl.f32(hl.sin(angle)))
    # Complex input element o as a (re, im) pair; x[0, o] is its real part and x[1, o] its imaginary part.
    X = hl.Func("X0")
    X[o] = (x[0, o], x[1, o])
    # Stockham radix-2 decimation in frequency: log2(n) out-of-place stages, output in natural order.
    s, m = 1, n // 2  # stride and half length of the sub-transforms of this stage (2*m*s == n)
    t = 0
    while m >= 1:
        t += 1
        Y = hl.Func(f"X{t}")
        q = o % s
        p = (o // s) // 2
        e = (o // s) % 2
        a = X[q + s * p]
        b = X[q + s * (p + m)]
        w = tw[p * s]  # exp(-2*pi*i*p/(2*m))
        dr, di = a[0] - b[0], a[1] - b[1]
        Y[o] = (hl.select(e == 0, a[0] + b[0], dr * w[0] - di * w[1]),
                hl.select(e == 0, a[1] + b[1], dr * w[1] + di * w[0]))
        X = Y
        s, m = 2 * s, m // 2
    # Output element k as row k of an (n, 2) array: out[0, k] is the real part, out[1, k] the imaginary part.
    out = hl.Func("out")
    out[c, k] = hl.select(c == 0, X[k][0], X[k][1])
    return out
```
