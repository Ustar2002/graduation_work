The distance matrix starts as a copy of W and is relaxed in place with the intermediate vertex k as the outermost reduction loop; for each k the rows are distributed over the threads and the j loop inside a row is vectorized.

```python
import halide as hl


def floyd_warshall(W, n):
    # Halide coordinates are reversed: W[i][j] is W[j, i] and D[i][j] is D[j, i].
    j, i = hl.Var("j"), hl.Var("i")
    D = hl.Func("D")
    D[j, i] = W[j, i]
    # In-place relaxation; the reduction loops run in order: k (outermost), then i, then j (innermost).
    # Row k cannot change while k is the intermediate vertex (D[k][k] stays 0), so it is skipped.
    r = hl.RDom([(0, n), (0, n), (0, n)], "r")  # r.x = j, r.y = i, r.z = k
    r.where(r.y != r.z)
    D[r.x, r.y] = hl.min(D[r.x, r.y], D[r.z, r.y] + D[r.x, r.z])

    # Schedule. For a fixed k, the task for row i writes only row i and reads only row i and row k, and row k
    # is not written while k is the intermediate vertex (r.y != r.z): the rows are independent, so the race
    # check (which cannot prove this) is overridden. Inside a row, j is split by 8 (guarded tail) and vectorized.
    jo, ji = hl.Var("jo"), hl.Var("ji")
    D.split(j, jo, ji, 8, hl.TailStrategy.GuardWithIf).vectorize(ji).parallel(i)
    rxo, rxi = hl.RVar("rxo"), hl.RVar("rxi")
    D.update(0).allow_race_conditions().split(r.x, rxo, rxi, 8, hl.TailStrategy.GuardWithIf).vectorize(rxi).parallel(r.y)
    return D
```
