The distances are relaxed in place in D, one intermediate vertex at a time; the row of the intermediate vertex does not change in its own step and is skipped.

```c
#include <stddef.h>
#include <string.h>

void floyd_warshall(const float* W, float* D, int n) {
    memcpy(D, W, (size_t)n * (size_t)n * sizeof(float));
    for (int k = 0; k < n; k++) {
        const float *dk = D + (size_t)k * n;
        for (int i = 0; i < n; i++) {
            if (i == k) continue; /* 행 k 는 스텝 k 에서 바뀌지 않는다(D[k][k] = 0) */
            float *restrict di = D + (size_t)i * n;
            const float dik = di[k];
            for (int j = 0; j < n; j++) {
                const float t = dik + dk[j];
                di[j] = t < di[j] ? t : di[j];
            }
        }
    }
}
```
