dist_out starts as a copy of dist_in; the loop over the intermediate vertex k stays sequential, and within one step every row is updated independently in parallel with `par_chunks_mut`. Row k is copied into a separate buffer before each step, so no task reads memory that another task writes.

```rust
use rayon::prelude::*;

#[no_mangle]
pub unsafe extern "C" fn floyd_warshall(dist_in: *const f32, dist_out: *mut f32, n: i32) {
    if n <= 0 {
        return;
    }
    let n = n as usize;
    let w = std::slice::from_raw_parts(dist_in, n * n);
    let d = std::slice::from_raw_parts_mut(dist_out, n * n);

    // Every element of dist_out is overwritten by the copy before the in-place relaxation.
    d.copy_from_slice(w);
    let mut rowk = vec![0.0f32; n];
    for k in 0..n {
        rowk.copy_from_slice(&d[k * n..(k + 1) * n]);
        let rk: &[f32] = &rowk;
        d.par_chunks_mut(n).for_each(|row| {
            let dik = row[k];
            for (x, &y) in row.iter_mut().zip(rk.iter()) {
                let s = dik + y;
                if s < *x {
                    *x = s;
                }
            }
        });
    }
}
```
