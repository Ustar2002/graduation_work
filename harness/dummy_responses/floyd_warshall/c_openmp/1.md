The outermost loop has the most iterations, so it is the one distributed over the threads: every thread takes a range of intermediate vertices k and relaxes the whole matrix through them, and the inner i/j loops run as in the sequential version.

```c
#include <string.h>

void floyd_warshall(const float* dist_in, float* dist_out, int n) {
    if (n <= 0) return;
    memcpy(dist_out, dist_in, (size_t)n * (size_t)n * sizeof(float));
#pragma omp parallel for schedule(static)
    for (int k = 0; k < n; k++) {
        const float *rowk = dist_out + (size_t)k * n;
        for (int i = 0; i < n; i++) {
            float *rowi = dist_out + (size_t)i * n;
            const float dik = rowi[k];
            for (int j = 0; j < n; j++) {
                float via = dik + rowk[j];
                if (via < rowi[j]) rowi[j] = via;
            }
        }
    }
}
```
