The loop over the intermediate vertex stays sequential; within one step the rows are independent, so the row loop is parallel. The row of the intermediate vertex is skipped, so no element is read by one thread while another writes it.

```c
#include <stddef.h>
#include <string.h>

void floyd_warshall(const float* W, float* D, int n) {
    memcpy(D, W, (size_t)n * (size_t)n * sizeof(float));
    for (int k = 0; k < n; k++) { /* 스텝 k 는 스텝 0..k-1 의 결과를 읽으므로 순차 */
        const float *dk = D + (size_t)k * n;
        /* 반복 i 는 행 i 만 쓰고 행 i·행 k 만 읽는다. 행 k 는 스텝 k 에서 바뀌지 않으므로(D[k][k] = 0) 건너뛴다 */
#pragma omp parallel for schedule(static)
        for (int i = 0; i < n; i++) {
            if (i == k) continue;
            float *restrict di = D + (size_t)i * n;
            const float dik = di[k];
            for (int j = 0; j < n; j++) {
                const float t = dik + dk[j];
                di[j] = t < di[j] ? t : di[j];
            }
        }
    }
}
```
