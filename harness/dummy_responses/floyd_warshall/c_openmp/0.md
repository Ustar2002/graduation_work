The k loop is sequential (each step depends on the previous one); within a step the rows i are independent, so they are split across threads and the inner j loop is left to the compiler's vectorizer. Row k itself never changes in step k (d[k][k] = 0), so it is skipped.

```c
#include <string.h>

void floyd_warshall(const float* dist_in, float* dist_out, int n) {
    if (n <= 0) return;
    size_t nn = (size_t)n * (size_t)n;
    memcpy(dist_out, dist_in, nn * sizeof(float));
    for (int k = 0; k < n; k++) {
        const float *rowk = dist_out + (size_t)k * n;
#pragma omp parallel for schedule(static)
        for (int i = 0; i < n; i++) {
            if (i == k) continue;
            float *rowi = dist_out + (size_t)i * n;
            const float dik = rowi[k];
            for (int j = 0; j < n; j++) {
                float via = dik + rowk[j];
                rowi[j] = via < rowi[j] ? via : rowi[j];
            }
        }
    }
}
```
