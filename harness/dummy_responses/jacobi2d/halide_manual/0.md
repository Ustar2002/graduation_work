Each of the T steps is its own Func that reads only the previous step: interior cells take the average of their four neighbours and every other cell copies its previous value. Every step is computed completely before the next one starts, with the rows distributed over the threads and each row vectorized.

```python
import halide as hl


def jacobi2d(src, H, W, T):
    # Halide coordinates are reversed: the cell in row y and column x is u[x, y].
    x, y = hl.Var("x"), hl.Var("y")
    interior = (x >= 1) & (x <= W - 2) & (y >= 1) & (y <= H - 2)
    # Step 0 is the input, clamped so that it is defined (and readable) outside the grid as well.
    u = hl.Func("u0")
    u[x, y] = hl.BoundaryConditions.repeat_edge(src)[x, y]
    steps = []
    for t in range(1, T + 1):
        nxt = hl.Func("u%d" % t)
        nxt[x, y] = hl.select(interior, 0.25 * (u[x, y - 1] + u[x, y + 1] + u[x - 1, y] + u[x + 1, y]), u[x, y])
        steps.append(nxt)
        u = nxt

    # Schedule: u0 stays inlined (it reads src directly); every step is one pass over its region,
    # rows in parallel and x vectorized. Each pass writes only its own buffer, so there is no race.
    for k, f in enumerate(steps):
        if k + 1 < len(steps):
            f.compute_root()
        f.vectorize(x, 8, hl.TailStrategy.GuardWithIf).parallel(y)
    return u
```
