Two buffers (`dst` and a temporary copy) are swapped each step, with the first target chosen so that the last step writes into `dst`; within a step the rows of the output are split across rayon threads.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

/// One Jacobi step: interior cells of `out` from `inp`. Boundary cells of `out` are not touched
/// (they already hold the input values).
fn step(inp: &[f32], out: &mut [f32], h: usize, w: usize) {
    // Each closure writes only its own row of `out` (disjoint chunks) and reads only `inp`.
    out.par_chunks_mut(w).enumerate().for_each(|(y, row)| {
        if y == 0 || y == h - 1 {
            return;
        }
        let up = &inp[(y - 1) * w..y * w];
        let mid = &inp[y * w..(y + 1) * w];
        let down = &inp[(y + 1) * w..(y + 2) * w];
        for x in 1..w - 1 {
            row[x] = 0.25 * (up[x] + down[x] + mid[x - 1] + mid[x + 1]);
        }
    });
}

#[no_mangle]
pub unsafe extern "C" fn jacobi2d(src: *const f32, dst: *mut f32, H: i32, W: i32, T: i32) {
    if H <= 0 || W <= 0 {
        return;
    }
    let (h, w) = (H as usize, W as usize);
    let n = h * w;
    let src = std::slice::from_raw_parts(src, n);
    let dst = std::slice::from_raw_parts_mut(dst, n);

    // Full copy, boundary included: the boundary never changes afterwards.
    dst.copy_from_slice(src);
    if T <= 0 || h < 3 || w < 3 {
        return;
    }
    let t_steps = T as usize;
    let mut tmp: Vec<f32> = if t_steps >= 2 { src.to_vec() } else { Vec::new() };

    for t in 0..t_steps {
        // Steps whose distance to the last step is even write into dst, so the last step does.
        let to_dst = (t_steps - 1 - t) % 2 == 0;
        if t == 0 {
            if to_dst {
                step(src, dst, h, w);
            } else {
                step(src, &mut tmp, h, w);
            }
        } else if to_dst {
            step(&tmp, dst, h, w);
        } else {
            step(dst, &mut tmp, h, w);
        }
    }
}
```
