Two buffers are swapped each step and the rows of every step are split across threads with `prange`. The order of the buffers is chosen so that the last step writes into `dst`; both buffers start as copies of `src`, so the boundary cells already hold their final values.

```python
import numpy as np
from numba import njit, prange


@njit("void(float32[:, ::1], float32[:, ::1], int64, int64)", parallel=True)
def _step(inp, out, H, W):
    # Iteration y writes only row y of out and reads only inp, so there are no races.
    for y in prange(1, H - 1):
        for x in range(1, W - 1):
            out[y, x] = np.float32(0.25) * (inp[y - 1, x] + inp[y + 1, x] + inp[y, x - 1] + inp[y, x + 1])


@njit("void(float32[:, ::1], float32[:, ::1], int64, int64, int64)")
def jacobi2d(src, dst, H, W, T):
    dst[:, :] = src
    if T <= 0:
        return
    if T == 1:
        _step(src, dst, H, W)
        return
    tmp = src.copy()
    # With an odd T the first step writes dst, with an even T it writes tmp; the last step always writes dst.
    if T % 2 == 1:
        _step(src, dst, H, W)
    else:
        _step(src, tmp, H, W)
    for t in range(1, T):
        if (T - t) % 2 == 1:
            _step(tmp, dst, H, W)
        else:
            _step(dst, tmp, H, W)
```
