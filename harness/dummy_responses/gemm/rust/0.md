The rows of `C` are independent, so they are distributed over the rayon thread pool with `par_chunks_mut`, one row per chunk; inside a row the i-k-j order keeps the innermost loop contiguous over `B` and `C` so the compiler can vectorize it. Each row is zeroed before accumulation, so the initial contents of `C` are ignored.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

#[no_mangle]
pub unsafe extern "C" fn gemm(A: *const f32, B: *const f32, C: *mut f32, M: i32, N: i32, K: i32) {
    if M <= 0 || N <= 0 || K <= 0 {
        return;
    }
    let (m, n, k) = (M as usize, N as usize, K as usize);
    let a = std::slice::from_raw_parts(A, m * k);
    let b = std::slice::from_raw_parts(B, k * n);
    let c = std::slice::from_raw_parts_mut(C, m * n);

    // Each closure writes only its own row of C (disjoint chunks), so there is no data race.
    c.par_chunks_mut(n).enumerate().for_each(|(i, c_row)| {
        c_row.iter_mut().for_each(|v| *v = 0.0);
        let a_row = &a[i * k..(i + 1) * k];
        for (kk, &aik) in a_row.iter().enumerate() {
            let b_row = &b[kk * n..(kk + 1) * n];
            for (cv, &bv) in c_row.iter_mut().zip(b_row) {
                *cv += aik * bv;
            }
        }
    });
}
```
