The keys are split into large chunks; each chunk is counted into its own private histogram in parallel, and the private histograms are then added bin by bin. Integer addition is exact, so the result does not depend on the number of threads.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

#[no_mangle]
pub unsafe extern "C" fn histogram(keys: *const i32, hist: *mut i32, n: i32, n_bins: i32) {
    let nb = n_bins.max(0) as usize;
    let len = n.max(0) as usize;
    let hist = std::slice::from_raw_parts_mut(hist, nb);
    let keys = std::slice::from_raw_parts(keys, len);

    // Each chunk is at least four times as long as the histogram, so the private
    // histograms together never take more memory than the keys themselves.
    let chunk = (1usize << 20).max(4 * nb).max(1);
    // Every closure only reads keys and writes its own freshly allocated vector:
    // no shared writes, no atomics needed.
    let total: Vec<u32> = keys
        .par_chunks(chunk)
        .map(|part| {
            let mut local = vec![0u32; nb];
            for &k in part {
                local[k as usize] += 1;
            }
            local
        })
        .reduce(
            || vec![0u32; nb],
            |mut a, b| {
                for (x, y) in a.iter_mut().zip(b.iter()) {
                    *x += *y;
                }
                a
            },
        );
    for (h, &c) in hist.iter_mut().zip(total.iter()) {
        *h = c as i32;
    }
}
```
