Each thread counts its share of `data` into a private histogram (one contiguous slice of a scratch array, so no two threads write the same cache lines), then the bins are summed across threads with the bin loop shared, which writes every element of `hist` exactly once.

```c
#include <stdlib.h>
#include <string.h>
#include <omp.h>

void histogram(const int* data, int* hist, int n, int nbins) {
    int nt = omp_get_max_threads();
    int *priv = calloc((size_t)nt * (size_t)nbins, sizeof(int));
    if (!priv) {
        memset(hist, 0, (size_t)nbins * sizeof(int));
        for (int i = 0; i < n; i++) hist[data[i]]++;
        return;
    }
#pragma omp parallel num_threads(nt)
    {
        int *local = priv + (size_t)omp_get_thread_num() * (size_t)nbins;
#pragma omp for schedule(static)
        for (int i = 0; i < n; i++) local[data[i]]++;
#pragma omp for schedule(static)
        for (int b = 0; b < nbins; b++) {
            int s = 0;
            for (int t = 0; t < nt; t++) s += priv[(size_t)t * (size_t)nbins + b];
            hist[b] = s;
        }
    }
    free(priv);
}
```
