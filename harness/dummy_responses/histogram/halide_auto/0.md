The keys are split into chunks that each count into their own row of a partial histogram, so no two chunks share a counter; the rows are then added up per bin.

```python
import halide as hl


def histogram(keys, n, n_bins):
    chunks = min(n, 64)
    size = (n + chunks - 1) // chunks

    b, c = hl.Var("b"), hl.Var("c")
    r = hl.RDom([(0, size)], "r")
    i = c * size + r.x
    key = hl.unsafe_promise_clamped(keys[hl.min(i, n - 1)], 0, n_bins - 1)  # keys are valid bin indices

    partial = hl.Func("partial")
    partial[b, c] = 0
    partial[key, c] += hl.select(i < n, 1, 0)

    rc = hl.RDom([(0, chunks)], "rc")
    hist = hl.Func("hist")
    hist[b] = 0
    hist[b] += partial[b, rc.x]
    return hist
```
