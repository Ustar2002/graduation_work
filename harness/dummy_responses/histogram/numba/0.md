Each thread counts its own contiguous chunk of `data` into a private row of counters, so no two threads ever write the same counter; a second parallel loop over the bins then adds the private rows and overwrites every element of `hist`.

```python
import numba
import numpy as np
from numba import njit, prange


@njit("void(int32[::1], int32[::1], int64, int64)", parallel=True)
def histogram(data, hist, n, nbins):
    nchunks = numba.get_num_threads()
    if nchunks > n:
        nchunks = n
    if nchunks < 1:
        nchunks = 1
    chunk = (n + nchunks - 1) // nchunks
    local = np.zeros((nchunks, nbins), dtype=np.int32)
    # Iteration c writes only row c of local: no shared counters, no race.
    for c in prange(nchunks):
        start = c * chunk
        stop = min(start + chunk, n)
        row = local[c]
        for i in range(start, stop):
            row[data[i]] += 1
    # Iteration b writes only hist[b].
    for b in prange(nbins):
        s = 0
        for c in range(nchunks):
            s += local[c, b]
        hist[b] = s
```
