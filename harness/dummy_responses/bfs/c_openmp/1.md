Level-synchronous BFS with the frontier loop parallelised: each thread scans its share of the current frontier, marks every still-unvisited neighbour with the next level and appends it to the next frontier. The frontier arrays are swapped after each level until no new vertex is found.

```c
#include <stdlib.h>

void bfs(const int* row_ptr, const int* col_idx, int* dist, int n, int nnz) {
    (void)nnz;
    if (n <= 0) return;
    for (int i = 0; i < n; i++) dist[i] = -1;
    dist[0] = 0;
    int *frontier = malloc((size_t)n * sizeof(int));
    int *next = malloc((size_t)n * sizeof(int));
    if (!frontier || !next) { free(frontier); free(next); return; }
    frontier[0] = 0;
    int size = 1;
    for (int level = 1; size > 0; level++) {
        int next_size = 0;
#pragma omp parallel for schedule(static)
        for (int f = 0; f < size; f++) {
            int u = frontier[f];
            for (int p = row_ptr[u]; p < row_ptr[u + 1]; p++) {
                int v = col_idx[p];
                if (dist[v] == -1) {
                    dist[v] = level;
                    next[next_size++] = v;
                }
            }
        }
        int *t = frontier; frontier = next; next = t;
        size = next_size;
    }
    free(frontier);
    free(next);
}
```
