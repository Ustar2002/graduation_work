Level-synchronous top-down BFS. The current frontier is split across threads with a dynamic `omp for`; a thread claims an unvisited neighbour with an atomic compare-and-swap on `dist` (-1 to level + 1), so each vertex is claimed exactly once, and appends it to a small thread-local buffer that is flushed into the shared next frontier through an atomic fetch-add on its size. The region ends with the implicit barrier, the two frontier arrays are swapped, and the loop stops when the frontier is empty; vertices never claimed keep the -1 written at the start.

```c
#include <stdlib.h>
#include <string.h>
#include <omp.h>

#define CHUNK 1024

void bfs(const int* row_ptr, const int* col_idx, int* dist, int n, int nnz) {
    (void)nnz;
    if (n <= 0) return;
#pragma omp parallel for schedule(static)
    for (int i = 0; i < n; i++) dist[i] = -1;
    dist[0] = 0;

    int *frontier = malloc((size_t)n * sizeof(int));
    int *next = malloc((size_t)n * sizeof(int));
    if (!frontier || !next) {
        free(frontier); free(next);
        for (int level = 0;; level++) {           /* queue-free fallback */
            int found = 0;
            for (int u = 0; u < n; u++) {
                if (dist[u] != level) continue;
                for (int p = row_ptr[u]; p < row_ptr[u + 1]; p++)
                    if (dist[col_idx[p]] == -1) { dist[col_idx[p]] = level + 1; found = 1; }
            }
            if (!found) return;
        }
    }
    frontier[0] = 0;
    int size = 1;
    for (int level = 1; size > 0; level++) {
        int next_size = 0;
#pragma omp parallel
        {
            int local[CHUNK];
            int cnt = 0;
#pragma omp for schedule(dynamic, 32) nowait
            for (int f = 0; f < size; f++) {
                int u = frontier[f];
                for (int p = row_ptr[u], end = row_ptr[u + 1]; p < end; p++) {
                    int v = col_idx[p];
                    if (__atomic_load_n(&dist[v], __ATOMIC_RELAXED) != -1) continue;
                    if (__sync_bool_compare_and_swap(&dist[v], -1, level)) {
                        local[cnt++] = v;
                        if (cnt == CHUNK) {
                            int base = __sync_fetch_and_add(&next_size, cnt);
                            memcpy(next + base, local, (size_t)cnt * sizeof(int));
                            cnt = 0;
                        }
                    }
                }
            }
            if (cnt > 0) {
                int base = __sync_fetch_and_add(&next_size, cnt);
                memcpy(next + base, local, (size_t)cnt * sizeof(int));
            }
        }
        int *t = frontier; frontier = next; next = t;
        size = next_size;
    }
    free(frontier);
    free(next);
}
```
