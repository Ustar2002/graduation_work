Level-synchronous sweep: for each level, all vertices are scanned in parallel and the vertices at the current level label their unvisited neighbours. Concurrent accesses to `dist` are atomic reads and writes, and every write in one sweep stores the same value, so the result does not depend on the thread schedule.

```c
void bfs(const int* row_ptr, const int* col_idx, int* dist, int n_vertices, int n_edges, int source) {
    (void)n_edges;
#pragma omp parallel for schedule(static)
    for (int v = 0; v < n_vertices; v++) dist[v] = -1; /* 도달 불가 정점은 이 값으로 남는다 */
    dist[source] = 0;
    for (int level = 0;; level++) {
        long found = 0; /* 이번 스윕에서 새로 거리를 준 횟수(같은 정점을 두 스레드가 세도 0 인지 아닌지만 쓴다) */
#pragma omp parallel for schedule(static) reduction(+ : found)
        for (int u = 0; u < n_vertices; u++) {
            int du;
#pragma omp atomic read
            du = dist[u];
            if (du != level) continue; /* 이번 스윕이 쓰는 값은 level + 1 뿐이라 이 판정은 실행 순서와 무관하다 */
            for (int p = row_ptr[u]; p < row_ptr[u + 1]; p++) {
                int v = col_idx[p];
                int dv;
#pragma omp atomic read
                dv = dist[v];
                if (dv < 0) {
#pragma omp atomic write
                    dist[v] = level + 1;
                    found++;
                }
            }
        }
        if (found == 0) break;
    }
}
```
