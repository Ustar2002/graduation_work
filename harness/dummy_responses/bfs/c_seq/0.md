A FIFO queue held in an array visits the vertices in order of increasing distance; every vertex enters the queue at most once.

```c
#include <stdlib.h>

void bfs(const int* row_ptr, const int* col_idx, int* dist, int n_vertices, int n_edges, int source) {
    (void)n_edges;
    for (int v = 0; v < n_vertices; v++) dist[v] = -1; /* 도달 불가 정점은 이 값으로 남는다 */
    int *queue = malloc((size_t)n_vertices * sizeof(int));
    if (!queue) return;
    int head = 0, tail = 0;
    dist[source] = 0;
    queue[tail++] = source;
    while (head < tail) {
        int u = queue[head++];
        int d = dist[u] + 1;
        for (int p = row_ptr[u]; p < row_ptr[u + 1]; p++) {
            int v = col_idx[p];
            if (dist[v] < 0) { /* 거리를 받는 순간 큐에 넣으므로 정점마다 한 번만 들어간다 */
                dist[v] = d;
                queue[tail++] = v;
            }
        }
    }
    free(queue);
}
```
