A standard queue-based BFS: `dist` is initialised to -1 (so unreachable vertices need no extra pass), vertex 0 is enqueued with distance 0, and each dequeued vertex relaxes its CSR range. Every vertex enters the queue at most once, so a plain array of n entries with head/tail indices is enough. If the queue cannot be allocated, a queue-free level sweep over all vertices is used instead.

```c
#include <stdlib.h>

void bfs(const int* row_ptr, const int* col_idx, int* dist, int n, int nnz) {
    (void)nnz;
    if (n <= 0) return;
    for (int i = 0; i < n; i++) dist[i] = -1;
    dist[0] = 0;
    int *queue = malloc((size_t)n * sizeof(int));
    if (!queue) {
        for (int level = 0;; level++) {
            int found = 0;
            for (int u = 0; u < n; u++) {
                if (dist[u] != level) continue;
                for (int p = row_ptr[u]; p < row_ptr[u + 1]; p++) {
                    int v = col_idx[p];
                    if (dist[v] == -1) { dist[v] = level + 1; found = 1; }
                }
            }
            if (!found) return;
        }
    }
    int head = 0, tail = 0;
    queue[tail++] = 0;
    while (head < tail) {
        int u = queue[head++];
        int d = dist[u] + 1;
        int end = row_ptr[u + 1];
        for (int p = row_ptr[u]; p < end; p++) {
            int v = col_idx[p];
            if (dist[v] == -1) {
                dist[v] = d;
                queue[tail++] = v;
            }
        }
    }
    free(queue);
}
```
