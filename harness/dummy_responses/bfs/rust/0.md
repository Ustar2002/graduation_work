This is a level-synchronous breadth-first search: the vertices of the current frontier are expanded in parallel with rayon, and a vertex joins the next frontier only through a successful atomic compare-and-swap of its distance from -1, so every vertex is claimed exactly once and the distances do not depend on thread timing. Every element of `dist` is first set to -1.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;
use std::sync::atomic::{AtomicI32, Ordering};

#[no_mangle]
pub unsafe extern "C" fn bfs(row_ptr: *const i32, col_idx: *const i32, dist: *mut i32, n_vertices: i32, n_edges: i32, source: i32) {
    if n_vertices <= 0 {
        return;
    }
    let nv = n_vertices as usize;
    let ne = if n_edges > 0 { n_edges as usize } else { 0 };
    let rp = std::slice::from_raw_parts(row_ptr, nv + 1);
    let ci = std::slice::from_raw_parts(col_idx, ne);
    // AtomicI32 has the same size and alignment as i32.
    let d = std::slice::from_raw_parts(dist as *const AtomicI32, nv);

    d.par_iter().for_each(|x| x.store(-1, Ordering::Relaxed));
    d[source as usize].store(0, Ordering::Relaxed);

    let mut frontier: Vec<i32> = vec![source];
    let mut level: i32 = 0;
    while !frontier.is_empty() {
        level += 1;
        frontier = frontier
            .par_chunks(256)
            .flat_map_iter(|chunk| {
                let mut next = Vec::new();
                for &u in chunk {
                    let u = u as usize;
                    for &v in &ci[rp[u] as usize..rp[u + 1] as usize] {
                        let dv = &d[v as usize];
                        if dv.load(Ordering::Relaxed) == -1
                            && dv.compare_exchange(-1, level, Ordering::Relaxed, Ordering::Relaxed).is_ok()
                        {
                            next.push(v);
                        }
                    }
                }
                next
            })
            .collect();
    }
}
```
