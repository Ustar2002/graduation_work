Level-synchronous breadth-first search from vertex 0. Numba has no atomic operations, so a frontier vertex cannot be claimed by a compare-and-swap; instead the frontier is cut into chunks that are expanded in parallel with `prange`, and every chunk writes the same value (the current level) into `dist` for each still unvisited neighbour, so a lost race never changes a distance. A neighbour found by two chunks is kept in the next frontier only by the chunk whose index is left in `owner` after the parallel loop, which removes the duplicates. Levels with few edges are expanded sequentially, and `dist` is set to -1 everywhere first.

```python
import numpy as np
from numba import njit, prange

CHUNK = 512        # frontier vertices per parallel chunk
SEQ_EDGES = 16384  # levels with at most this many edges are expanded sequentially


@njit("void(int32[::1], int32[::1], int32[::1], int64, int64)", parallel=True)
def bfs(row_ptr, col_idx, dist, n, nnz):
    for v in prange(n):
        dist[v] = -1
    dist[0] = 0

    frontier = np.empty(n, dtype=np.int32)
    nxt = np.empty(n, dtype=np.int32)
    owner = np.empty(n, dtype=np.int32)
    frontier[0] = 0
    fsize = 1
    level = 0
    while fsize > 0:
        level += 1
        nch = (fsize + CHUNK - 1) // CHUNK

        # Number of out-edges of each chunk, and their offsets in the candidate buffer.
        edges = np.empty(nch, dtype=np.int64)
        for c in prange(nch):
            lo0 = c * CHUNK
            hi0 = min(lo0 + CHUNK, fsize)
            s0 = 0
            for f0 in range(lo0, hi0):
                u0 = frontier[f0]
                s0 += row_ptr[u0 + 1] - row_ptr[u0]
            edges[c] = s0
        off = np.empty(nch + 1, dtype=np.int64)
        off[0] = 0
        for c in range(nch):
            off[c + 1] = off[c] + edges[c]
        total = off[nch]

        if total <= SEQ_EDGES:
            # Small level: plain sequential expansion, no duplicates possible.
            nsize = 0
            for f1 in range(fsize):
                u1 = frontier[f1]
                for p1 in range(row_ptr[u1], row_ptr[u1 + 1]):
                    w1 = col_idx[p1]
                    if dist[w1] == -1:
                        dist[w1] = level
                        nxt[nsize] = w1
                        nsize += 1
        else:
            cand = np.empty(total, dtype=np.int32)
            kept = np.empty(nch, dtype=np.int64)
            # Expansion: concurrent writes to dist[w] all store the same level.
            for c in prange(nch):
                lo2 = c * CHUNK
                hi2 = min(lo2 + CHUNK, fsize)
                base2 = off[c]
                k2 = 0
                for f2 in range(lo2, hi2):
                    u2 = frontier[f2]
                    for p2 in range(row_ptr[u2], row_ptr[u2 + 1]):
                        w2 = col_idx[p2]
                        if dist[w2] == -1:
                            dist[w2] = level
                            owner[w2] = c
                            cand[base2 + k2] = w2
                            k2 += 1
                kept[c] = k2
            # Deduplication: after the loop, owner[w] names exactly one of the chunks that found w.
            for c in prange(nch):
                base3 = off[c]
                k3 = 0
                for t3 in range(kept[c]):
                    w3 = cand[base3 + t3]
                    if owner[w3] == c:
                        cand[base3 + k3] = w3
                        k3 += 1
                kept[c] = k3
            out = np.empty(nch + 1, dtype=np.int64)
            out[0] = 0
            for c in range(nch):
                out[c + 1] = out[c] + kept[c]
            for c in prange(nch):
                base4 = off[c]
                dst4 = out[c]
                for t4 in range(kept[c]):
                    nxt[dst4 + t4] = cand[base4 + t4]
            nsize = out[nch]

        frontier, nxt = nxt, frontier
        fsize = nsize
```
