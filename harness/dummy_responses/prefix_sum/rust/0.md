This is a two-pass block scan: the block sums are computed in parallel, a short sequential scan turns them into block offsets, and each block then writes its own running totals starting from its offset in parallel.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

#[no_mangle]
pub unsafe extern "C" fn prefix_sum(x: *const i32, out: *mut i32, n: i32) {
    if n <= 0 {
        return;
    }
    let len = n as usize;
    let x = std::slice::from_raw_parts(x, len);
    let out = std::slice::from_raw_parts_mut(out, len);

    const BLOCK: usize = 1 << 16;
    // Pass 1: each closure only reads its block of x and returns the block sum.
    let sums: Vec<i32> = x.par_chunks(BLOCK).map(|b| b.iter().sum::<i32>()).collect();

    // Exclusive scan of the block sums (one value per block, done sequentially).
    let mut offsets = Vec::with_capacity(sums.len());
    let mut run = 0i32;
    for s in &sums {
        offsets.push(run);
        run += *s;
    }

    // Pass 2: each closure writes only its own disjoint block of out.
    out.par_chunks_mut(BLOCK)
        .zip(x.par_chunks(BLOCK))
        .zip(offsets.par_iter())
        .for_each(|((ob, xb), &off)| {
            let mut acc = off;
            for (o, &v) in ob.iter_mut().zip(xb.iter()) {
                acc += v;
                *o = acc;
            }
        });
}
```
