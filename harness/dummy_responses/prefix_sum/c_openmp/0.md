The array is split into one contiguous block per thread and processed in two passes: the first pass sums each block, a short serial step turns the block sums into the starting total of every block, and the second pass computes the running totals of each block from its starting total.

```c
#include <omp.h>
#include <stdlib.h>

void prefix_sum(const int* x, int* out, int n) {
    int nb = omp_get_max_threads();
    if (nb > n) nb = n;
    int *start = nb > 1 ? malloc(((size_t)nb + 1) * sizeof(int)) : NULL;
    if (!start) {
        int s = 0;
        for (int i = 0; i < n; i++) {
            s += x[i];
            out[i] = s;
        }
        return;
    }
#pragma omp parallel for schedule(static)
    for (int b = 0; b < nb; b++) {
        size_t lo = (size_t)n * b / nb, hi = (size_t)n * (b + 1) / nb;
        int s = 0;
        for (size_t i = lo; i < hi; i++) s += x[i];
        start[b + 1] = s;
    }
    start[0] = 0;
    for (int b = 1; b <= nb; b++) start[b] += start[b - 1];
#pragma omp parallel for schedule(static)
    for (int b = 0; b < nb; b++) {
        size_t lo = (size_t)n * b / nb, hi = (size_t)n * (b + 1) / nb;
        int s = start[b];
        for (size_t i = lo; i < hi; i++) {
            s += x[i];
            out[i] = s;
        }
    }
    free(start);
}
```
