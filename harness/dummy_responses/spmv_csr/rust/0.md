The rows of `y` are independent, so blocks of rows are distributed over the rayon thread pool; each row is accumulated in double precision.

```rust
#![allow(non_snake_case)]
use rayon::prelude::*;

#[no_mangle]
pub unsafe extern "C" fn spmv_csr(row_ptr: *const i32, col_idx: *const i32, vals: *const f32, x: *const f32, y: *mut f32, n_rows: i32, n_cols: i32, nnz: i32) {
    if n_rows <= 0 {
        return;
    }
    let nr = n_rows as usize;
    let nc = n_cols.max(0) as usize;
    let nz = nnz.max(0) as usize;
    let row_ptr = std::slice::from_raw_parts(row_ptr, nr + 1);
    let col_idx: &[i32] = if nz > 0 { std::slice::from_raw_parts(col_idx, nz) } else { &[] };
    let vals: &[f32] = if nz > 0 { std::slice::from_raw_parts(vals, nz) } else { &[] };
    let x: &[f32] = if nc > 0 { std::slice::from_raw_parts(x, nc) } else { &[] };
    let y = std::slice::from_raw_parts_mut(y, nr);

    const ROWS_PER_CHUNK: usize = 1024;
    // Each closure writes only its own block of y (disjoint chunks); all other arrays are only read.
    y.par_chunks_mut(ROWS_PER_CHUNK).enumerate().for_each(|(chunk, y_block)| {
        let first = chunk * ROWS_PER_CHUNK;
        for (r, out) in y_block.iter_mut().enumerate() {
            let i = first + r;
            let start = row_ptr[i] as usize;
            let end = row_ptr[i + 1] as usize;
            let mut sum = 0.0f64;
            for p in start..end {
                sum += vals[p] as f64 * x[col_idx[p] as usize] as f64;
            }
            *out = sum as f32;
        }
    });
}
```
