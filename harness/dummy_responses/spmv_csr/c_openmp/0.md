Rows are independent, so the row loop is parallel.

```c
/* spmv_csr 성능 기준선: 손으로 쓴 C + OpenMP (CLAUDE.md §4). 행 단위 정적 분할. */
void spmv_csr(const int* row_ptr, const int* col_idx, const float* vals, const float* x, float* y,
              int n_rows, int n_cols, int nnz) {
    (void)n_cols; (void)nnz;
#pragma omp parallel for schedule(static)
    for (int i = 0; i < n_rows; i++) {
        float s = 0.0f;
        for (int p = row_ptr[i]; p < row_ptr[i + 1]; p++) s += vals[p] * x[col_idx[p]];
        y[i] = s;
    }
}
```
