The products vals[p] * x[col_idx[p]] are accumulated into a running sum S in double precision (S[0] = 0, S[p+1] = S[p] + product p), and each row's result is the difference of the running sum at the row's end and start, so an empty row gives exactly 0. The running sum is one sequential pass computed once; the rows are then computed in parallel blocks and vectorized.

```python
import halide as hl


def spmv_csr(row_ptr, col_idx, vals, x, n_rows, n_cols, nnz):
    i, p = hl.Var("i"), hl.Var("p")
    # Running sum of the products over all stored entries, in double precision; S is read at 0 .. nnz.
    S = hl.Func("S")
    S[p] = hl.f64(0)
    if nnz > 0:
        q = hl.RDom([(0, nnz)], "q")
        S[q + 1] = S[q] + hl.f64(vals[q]) * hl.f64(x[hl.clamp(col_idx[q], 0, n_cols - 1)])
    # Row i owns the entries row_ptr[i] .. row_ptr[i+1]-1 (the clamps only state the valid range of a CSR input).
    y = hl.Func("y")
    y[i] = hl.f32(S[hl.clamp(row_ptr[i + 1], 0, nnz)] - S[hl.clamp(row_ptr[i], 0, nnz)])

    # Schedule: S once over its whole range (the scan itself stays sequential: each step reads the previous one);
    # y in parallel blocks of 4096 rows, each block vectorized; every row writes only y[i].
    S.compute_root().vectorize(p, 8, hl.TailStrategy.GuardWithIf)
    if nnz > 0:
        S.update(0).unscheduled()
    io, ii = hl.Var("io"), hl.Var("ii")
    y.split(i, io, ii, 4096, hl.TailStrategy.GuardWithIf).parallel(io).vectorize(ii, 8, hl.TailStrategy.GuardWithIf)
    return y
```
